export type LinkStatus = "connecting" | "open" | "reconnecting" | "failed" | "replaced" | "removed";
export type BannerMode = "hidden" | "reconnecting" | "restored";
export interface LinkState { status: LinkStatus; banner: BannerMode; visible: boolean; failingSince: number | null; attempts: number;
  epoch: number; hasView: boolean; restoredUntil: number | null; removedReason: RemovedReason | null }
export type LinkEvent =
  | { type: "open"; now: number } | { type: "state"; now: number } | { type: "close"; code: number; reason: string; now: number }
  | { type: "visibility"; visible: boolean; now: number } | { type: "tick"; now: number } | { type: "retry"; now: number } | { type: "roomGone"; now: number };
export const FAIL_AFTER_MS = 30_000; export const RESTORED_MS = 2_000; export const STALE_CHECK_AFTER = 3;
export function initialLinkState(now: number, visible: boolean): LinkState;
export function linkReducer(s: LinkState, e: LinkEvent): LinkState;
export function retryDelayMs(attempts: number, rand01: number): number;   // min(5000, 500·2^(attempts−1)) + floor(rand01·250)

export function useRoomConnection(code: string, token: string, options?: {

## 실행 메모 — 웨이브 5 끝 인계(push)

이 계획은 웨이브 5(01-09, 01-10, 01-11)의 마지막 번호다. CLAUDE.md 작업 규칙(Mac·집 Windows PC 를 오가므로 작업을 마칠 때마다 HANDOFF 갱신·push)에 따라 아래를 한다. 중간 push 는 인계 편의이고 이 계획의 완료 조건이 아니다 — 최종 push 검증은 01-14 Task 3 이 한다. HANDOFF.md 는 SUMMARY 처럼 계획 범위(files_modified) 밖의 인계 파일이며, 아래 1번 조건(본 작업 트리의 main)일 때만 고치므로 병렬 워크트리와 겹치지 않는다.
1. SUMMARY 를 쓰기 전에 조건을 확인한다: `git rev-parse --abbrev-ref HEAD` 가 `main` 이고 `git rev-parse --git-dir` 와 `git rev-parse --git-common-dir` 의 결과가 같다(연결된 워크트리가 아닌 저장소 본 작업 트리), 그리고 같은 웨이브의 01-09·01-10 SUMMARY 가 이미 커밋돼 있다. 하나라도 아니면(병렬 워크트리 실행 등) HANDOFF 를 고치지 않고 push 하지 않으며, SUMMARY 끝에 `웨이브 5 push 미룸 — 병합 뒤 main 에서 HANDOFF 한 줄 갱신과 git push origin main 필요` 한 줄을 적는다(다음 웨이브 끝 push 나 01-14 가 함께 올린다).
2. 조건이 맞으면 SUMMARY 커밋 뒤 HANDOFF.md "현재 상태" 끝에 한 줄을 더한다(같은 1페이즈 실행 진행 줄이 이미 있으면 그 줄을 바꾼다): `- **1페이즈 실행 중 — 웨이브 5 까지 완료** ({오늘 날짜}): 01-01~01-11 완료, 다음 웨이브 6(01-12). 상세는 각 SUMMARY`. "다음 할 일" 첫 항목은 `/gsd-execute-phase 1` 로 이어서 실행(웨이브 6(01-12) 부터)으로 둔다. 비밀값·토큰·계정 ID 는 적지 않는다. `docs(01-11): 인계 — 1페이즈 웨이브 5 완료` 로 HANDOFF.md 만 커밋한다.
3. `git remote get-url origin` 이 `git@github.com:dhkim1522/minifia.git` 일 때만 `git push origin main`. 저장소에 걸린 개인 SSH 설정(~/.gitconfig-personal)을 그대로 쓴다. 전역 ~/.gitconfig 수정, `gh auth setup-git`, HTTPS 자격 증명 설정, force push, 스스로 rebase 는 하지 않는다.
4. push 가 거부되거나(non-fast-forward — 다른 PC 의 커밋) 인증이 실패하면 고치려 하지 말고 그 이유를 SUMMARY 끝에 한 줄 덧붙여 커밋한 뒤 계획을 끝낸다(다음 웨이브 끝이나 01-14 가 다시 올린다).
