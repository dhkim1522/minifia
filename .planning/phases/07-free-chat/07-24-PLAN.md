export function blendRemaining(elapsedMs: number): number; // 0 이하 → 1, SEG_BLEND_MS 이상 → 0(정확히), 사이 → 1 − p²(3 − 2p), p = elapsedMs / SEG_BLEND_MS

export function stepYaw(cur: number, target: number, dtSec: number): number; // 짧은 쪽으로 최대 YAW_TURN_RATE × dtSec, 남은 각이 그 이하면 target 을 (−π, π] 로 정규화해 그대로
export type OwnMoveEvent = "answered" | "reset" | null;
export interface OwnMoves {
  sent(pred: Segment | null): void;                                   // move 를 보낸 직후: 대기 +1, 예측 = pred(null 이면 예측 없음)
  observe(seq: number, server: Segment | undefined): OwnMoveEvent;    // 프레임마다 moveSeqOf(내 자리)·segOf(내 자리)

  reset(seq: number): void;                                           // 새 연결(epoch 변경): 대기·예측을 비우고 기준 seq 를 다시 잡는다
  effective(server: Segment | undefined): Segment | undefined;        // 그릴 구간: 예측 중이면 예측, 아니면 server
  pending(): number;
}
export function createOwnMoves(initialSeq: number): OwnMoves;

// src/world/store.ts (WorldStore 메서드)
moveSeqOf(seat: number): number; // 이 연결의 snap 뒤 그 자리에 적용된 move 수. 없으면 0

observe 규칙(시험이 고정한다):
1. seq < 기준 seq → 기준 = seq, 대기 0·예측 null, "reset".
2. seq > 기준 seq → d = seq − 기준, 기준 = seq. 대기가 0 이었으면 null(내가 보낸 것의 응답이 아님). 아니면 대기 = max(0, 대기 − d). 대기가 0 이 되었거나, 예측이 있고 server 의 마지막 두 수(끝점)가 예측의 마지막 두 수와 같으면 대기 0·예측 null, "answered". 그 밖에는 null(앞선 탭의 응답 — 예측을 끊지 않는다).
3. seq == 기준 → null.

## 참고 자료 (design-refs)

참고 이미지·에셋 경로 없음. 순수 계산 모듈이라 화면 모양이 없다. 섞기 시간(150ms)·방향 회전(반 바퀴 120ms)은 결정 0059 의 "100~150ms·짧게" 안에서 고른 값이다(Claude 재량, 아래 근거).

재량 선택과 근거:
- SEG_BLEND_MS = 150: 예측이 서버보다 `MOVE_V × RTT/2`(RTT 155ms 에서 0.14 단위) 앞서므로 교정 때 그만큼 뒤로 줄여야 한다. 범위의 위쪽 끝을 써 교정 중 감속을 가장 약하게 한다. 감쇠는 smoothstep 이라 최대 감소율이 평균의 1.5배(0.14 단위 기준 약 1.4 단위/초)로 걷는 속도 1.8 보다 작아 RTT 300ms 까지는 화면에서 뒤로 걷지 않는다.
- BLEND_SNAP_DIST = 3: 실제로 생기는 차이(RTT 앞섬 < 1, 응답 없음 되돌림 ≤ 2.7)는 모두 섞고, 그보다 큰 순간이동(재배치·시험 주입)은 섞지 않는다.
- YAW_TURN_RATE = π/0.12: 90° 전환이 약 60ms(60Hz 4프레임), 반 바퀴가 120ms 다.
